This document provides an overview of the polygenic score (PGS) and neuroimaging data available in the Brazilian High-Risk Cohort for Mental Health Conditions (BHRC). The goal of this resource is to describe the available genetic risk scores and neuroimaging measures generated for participants in the cohort, as well as the general structure of these datasets.

These READMEs are intended to facilitate the use of these datasets by providing information about the variables included, the processing steps applied, and the general organization of the data files.

calc_overlap <- function(df, var_prefix, waves = c("w0","w1","w2","w3", "overall")) {

  map_dfr(waves, function(w) {

    neuro_col <- paste0("genotyping_", w)
    var_col   <- paste0(var_prefix, "_", w)

    if(!var_col %in% names(df)) {
      return(tibble(wave = w, n = NA))
    }

    n_neuro <- sum(df[[neuro_col]] == 1, na.rm = TRUE)
    n_overlap <- sum(df[[neuro_col]] == 1 & df[[var_col]] == 1, na.rm = TRUE)

    tibble(
      wave = w,
      value = sprintf("%d (%.1f%%)", n_overlap, 100*n_overlap/n_neuro)
    )

  }) %>%
  pivot_wider(names_from = wave, values_from = value)
}

calc_overlap <- function(df, var_prefix, waves = c("w0","w1","w2","w3", "overall")) {

  map_dfr(waves, function(w) {

    neuro_col <- paste0("genotyping_", w)
    var_col   <- paste0(var_prefix, "_", w)

    if(!var_col %in% names(df)) {
      return(tibble(wave = w, n = NA))
    }

    n_neuro <- sum(df[[neuro_col]] == 1, na.rm = TRUE)
    n_overlap <- sum(df[[neuro_col]] == 1 & df[[var_col]] == 1, na.rm = TRUE)

    tibble(
      wave = w,
      value = sprintf("%d (%.1f%%)", n_overlap, 100*n_overlap/n_neuro)
    )

  }) %>%
  pivot_wider(names_from = wave, values_from = value)
}
